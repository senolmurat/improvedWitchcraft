			base = 0

# Hex Enemy, step 3 - the hex lands (hidden, weeks after witchimproved.1002)
# Root = the hexer. Reads scope:target, scope:hex_type, scope:hex_result, scope:hex_helper (optional).
witchimproved.1003 = {
	hidden = yes

	immediate = {
		# The victim died, or lost the capital the pestilence was meant for
		if = {
			limit = {
				OR = {
					scope:target = { is_alive = no }
					AND = {
						scope:hex_type = flag:pestilence
						scope:target = { NOT = { exists = capital_province } }
					}
				}
			}
			send_interface_toast = {
				title = witchimproved.1003.fizzle.t
				left_icon = scope:target
			}
		}
		else = {
			if = {
				limit = { scope:hex_result = flag:success }
				if = {
					limit = { scope:hex_type = flag:despair }
					send_interface_toast = {
						title = witchimproved.1003.success.t
						left_icon = scope:target
						show_as_tooltip = {
							scope:target = { witch_hex_despair_effect = yes }
						}
					}
					scope:target = {
						send_interface_toast = {
							title = witchimproved.1003.target.despair.t
							witch_hex_despair_effect = yes
						}
						# Now and then the fear takes root for good
						hidden_effect = {
							random = {
								chance = 5
								if = {
									limit = {
										NOR = {
											has_trait = paranoid
											has_trait = trusting
										}
									}
									add_trait = paranoid
								}
							}
						}
					}
				}
				else_if = {
					limit = { scope:hex_type = flag:sickness }
					send_interface_toast = {
						title = witchimproved.1003.success.t
						left_icon = scope:target
						show_as_tooltip = {
							scope:target = { witch_hex_sickness_effect = yes }
						}
					}
					scope:target = {
						send_interface_toast = {
							title = witchimproved.1003.target.sickness.t
							witch_hex_sickness_effect = yes
						}
						# The gift may also make them properly ill, more often when a skilled hand prepared it
						hidden_effect = {
							if = {
								limit = { exists = scope:hex_helper }
								random = {
									chance = 45
									if = {
										limit = { can_contract_disease_trigger = { DISEASE = ill } }
										contract_disease_effect = { DISEASE = ill TREATMENT_EVENT = yes }
									}
								}
							}
							else = {
								random = {
									chance = 30
									if = {
										limit = { can_contract_disease_trigger = { DISEASE = ill } }
										contract_disease_effect = { DISEASE = ill TREATMENT_EVENT = yes }
									}
								}
							}
						}
					}
				}
				else = {
					send_interface_toast = {
						title = witchimproved.1003.success.t
						left_icon = scope:target
						scope:target.capital_province = { witch_hex_epidemic_outbreak_effect = yes }
					}
				}
				hidden_effect = {
					witch_hex_murder_scheme_bonus_effect = yes
				}
			}
			else_if = {
				limit = { scope:hex_result = flag:backfire }
				send_interface_toast = {
					title = witchimproved.1003.backfire.t
					left_icon = scope:target
					witch_hex_backfire_effect = yes
				}
			}
			else_if = {
				limit = { scope:hex_result = flag:fizzle }
				send_interface_toast = {
					title = witchimproved.1003.fizzle.t
					left_icon = scope:target
				}
			}

			# Does the victim learn who was behind it?
			random = {
				chance = witch_hex_discovery_chance_value
				witch_hex_discovered_effect = yes
			}
		}
	}
}
