# Brew option of the Grand Rite (category rite_option_brew).
# Scope: the host. Run from the ritual phase on_end, so scope:activity and scope:host exist.
# Saves scope:brew_helper (an attending herbalist or court physician, if any) for the event portraits.
witch_improved_brew_outcome_effect = {
	scope:activity = {
		random_attending_character = {
			limit = {
				NOT = { this = scope:host }
				witch_improved_brew_helper_trigger = yes
			}
			save_scope_as = brew_helper
		}
	}
	if = {
		limit = {
			scope:activity = {
				has_activity_option = { category = rite_option_brew option = rite_brew_tea }
			}
		}
		trigger_event = witchimproved.1010
	}
	else_if = {
		limit = {
			scope:activity = {
				has_activity_option = { category = rite_option_brew option = rite_brew_strong }
			}
		}
		# Risky: a sick host can be cured, get a small boost, or get worse. A healthy host can also have a bad trip.
		if = {
			limit = { has_treatable_disease_trigger = yes }
			random_list = {
				40 = { trigger_event = witchimproved.1011 } # cured
				30 = { trigger_event = witchimproved.1013 } # not cured, small health boost
				30 = { trigger_event = witchimproved.1014 } # symptoms worsen
			}
		}
		else = {
			random_list = {
				85 = { trigger_event = witchimproved.1012 } # medium health boost
				15 = { trigger_event = witchimproved.1014 } # bad trip
			}
		}
	}
}

# Flow: witch.3014 (after) -> witchimproved.1005 (victim) -> witchimproved.1002 (hex) -> witchimproved.1003 (lands weeks later)
# Saved scopes carried through that chain: hexer, target, hex_helper (optional), hex_type, hex_result

# Builds the victim list for witchimproved.1005: murder scheme targets first, then rivals.
# Scope: the hexer. Saves scope:hexer, scope:hex_candidate_1..4 and scope:hex_has_passed_over (flag:yes/no).
# Victims the player passed over with "someone else" carry the timed flag witch_hex_passed_over.
witch_hex_find_candidates_effect = {
	save_scope_as = hexer
	# The event can fire again, so start clean
	if = {
		limit = { exists = scope:hex_candidate_1 }
		clear_saved_scope = hex_candidate_1
	}
	if = {
		limit = { exists = scope:hex_candidate_2 }
		clear_saved_scope = hex_candidate_2
	}
	if = {
		limit = { exists = scope:hex_candidate_3 }
		clear_saved_scope = hex_candidate_3
	}
	if = {
		limit = { exists = scope:hex_candidate_4 }
		clear_saved_scope = hex_candidate_4
	}
	if = {
		limit = {
			OR = {
				any_relation = {
					type = rival
					has_character_flag = witch_hex_passed_over
				}
				any_scheme = {
					type = murder
					scheme_target_character = { has_character_flag = witch_hex_passed_over }
				}
			}
		}
		save_scope_value_as = { name = hex_has_passed_over value = flag:yes }
	}
	else = {
		save_scope_value_as = { name = hex_has_passed_over value = flag:no }
	}

	every_scheme = {
		type = murder
		scheme_target_character = {
			if = {
				limit = {
					is_alive = yes
					NOT = { has_character_flag = witch_hex_passed_over }
					NOT = { is_in_list = hex_candidates }
				}
				add_to_temporary_list = hex_candidates
			}
		}
	}
	every_relation = {
		type = rival
		limit = {
			NOT = { has_character_flag = witch_hex_passed_over }
			NOT = { is_in_list = hex_candidates }
		}
		add_to_temporary_list = hex_candidates
	}
	every_in_list = {
		list = hex_candidates
		if = {
			limit = { NOT = { exists = scope:hex_candidate_1 } }
			save_scope_as = hex_candidate_1
		}
		else_if = {
			limit = { NOT = { exists = scope:hex_candidate_2 } }
			save_scope_as = hex_candidate_2
		}
		else_if = {
			limit = { NOT = { exists = scope:hex_candidate_3 } }
			save_scope_as = hex_candidate_3
		}
		else_if = {
			limit = { NOT = { exists = scope:hex_candidate_4 } }
			save_scope_as = hex_candidate_4
		}
	}
}

# Scope: the hexer
witch_hex_clear_passed_over_effect = {
	every_relation = {
		type = rival
		limit = { has_character_flag = witch_hex_passed_over }
		remove_character_flag = witch_hex_passed_over
	}
	every_scheme = {
		type = murder
		scheme_target_character = {
			if = {
				limit = { has_character_flag = witch_hex_passed_over }
				remove_character_flag = witch_hex_passed_over
			}
		}
	}
}

# Despair: the victim hears that a coven has cursed them, and fear does the rest.
# Scope: the victim. Reads scope:hexer. Deterministic, so it can also be shown as a tooltip.
witch_hex_despair_effect = {
	if = {
		limit = {
			OR = {
				has_trait = brave
				has_trait = cynical
			}
		}
		add_stress = medium_stress_gain
	}
	else_if = {
		limit = {
			OR = {
				has_trait = craven
				has_trait = paranoid
				has_trait = zealous
			}
		}
		add_stress = massive_stress_gain
	}
	else = {
		add_stress = major_stress_gain
	}
	if = {
		limit = { scope:hexer = { witch_improved_is_master_witch_trigger = yes } }
		add_character_modifier = {
			modifier = witch_hex_dread_modifier
			years = 4
		}
	}
	else_if = {
		limit = { scope:hexer = { witch_improved_is_adept_witch_trigger = yes } }
		add_character_modifier = {
			modifier = witch_hex_dread_modifier
			years = 3
		}
	}
	else = {
		add_character_modifier = {
			modifier = witch_hex_dread_modifier
			years = 2
		}
	}
}
