grand_rite_good_outcome = {
	scope:activity = {
		every_attending_character = {
			limit = { this != root }
			custom = every_grand_rite_participant
			add_opinion = {
				target = root
				modifier = impressed_opinion
				opinion = grand_rite_good_outcome_opinion_value
			}
		}
	}
}

grand_rite_poor_outcome = {
	scope:activity = {
		every_attending_character = {
			limit = { this != root }
			custom = every_grand_rite_participant
			add_opinion = {
				target = root
				modifier = disappointed_opinion
				opinion = grand_rite_bad_outcome_opinion_value
			}
		}
	}
	add_stress = minor_stress_gain
}

hosted_successful_witch_coven_new_effect = {
	# Piety gain from Tenets.
	if = {
		limit = {
			rite = { rite_has_parameter = piety_from_witch_coven_active }
		}
		add_piety = grand_rite_piety_reward_value
	}

	# prestige gain from culture
	if = {
		limit = {
			culture = { has_cultural_parameter = witch_trait_more_valued }
		}
		add_prestige = grand_rite_prestige_reward_value
	}

	witch_improved_witchcraft_trait_xp_gain_effect = yes
}

attended_successful_witch_coven_new_effect = {# witch trait xp gain
	witch_improved_witchcraft_trait_xp_gain_effect = yes
}

	save_temporary_scope_as = participant
	scope:activity = {
		add_activity_log_entry = {

			tags = { completed }

				attended_successful_witch_coven_new_effect = yes
			}
		}
	}
}

# Brew option of the Grand Rite (category rite_option_brew).
# Scope: the host. Run from the ritual phase on_end, so scope:activity and scope:host exist.
# Saves scope:brew_helper (an attending herbalist or court physician, if any) for the event portraits.
witch_improved_brew_outcome_effect = {
	scope:activity = {
		random_attending_character = {
			limit = {
				NOT = { this = scope:host }
				witch_improved_brew_helper_trigger = yes
			}
			save_scope_as = brew_helper
		}
	}
	if = {
		limit = {
			scope:activity = {
				has_activity_option = { category = rite_option_brew option = rite_brew_tea }
			}
		}
		trigger_event = witchimproved.1010
	}
	else_if = {
		limit = {
			scope:activity = {
				has_activity_option = { category = rite_option_brew option = rite_brew_strong }
			}
		}
		# Risky: a sick host can be cured, get a small boost, or get worse. A healthy host can also have a bad trip.
		if = {
			limit = { has_treatable_disease_trigger = yes }
			random_list = {
				40 = { trigger_event = witchimproved.1011 } # cured
				30 = { trigger_event = witchimproved.1013 } # not cured, small health boost
				30 = { trigger_event = witchimproved.1014 } # symptoms worsen
			}
		}
		else = {
			random_list = {
				85 = { trigger_event = witchimproved.1012 } # medium health boost
				15 = { trigger_event = witchimproved.1014 } # bad trip
			}
		}
	}
}

# Adds lifestyle XP to the character's current lifestyle focus. Scope: character. XP = amount.
# The wanderer lifestyle gets double, as it did in the original event.
witch_improved_add_current_lifestyle_xp_effect = {
	if = {
		limit = { has_lifestyle = diplomacy_lifestyle }
		add_diplomacy_lifestyle_xp = $XP$
	}
	else_if = {
		limit = { has_lifestyle = martial_lifestyle }
		add_martial_lifestyle_xp = $XP$
	}
	else_if = {
		limit = { has_lifestyle = stewardship_lifestyle }
		add_stewardship_lifestyle_xp = $XP$
	}
	else_if = {
		limit = { has_lifestyle = intrigue_lifestyle }
		add_intrigue_lifestyle_xp = $XP$
	}
	else_if = {
		limit = { has_lifestyle = learning_lifestyle }
		add_learning_lifestyle_xp = $XP$
	}
	else_if = {
		limit = { has_lifestyle = wanderer_lifestyle }
		add_wanderer_lifestyle_xp = {
			value = $XP$
			multiply = 2
		}
	}
}

# Seek Forbidden Knowledge, risky option succeeded. Scope: the host.
witch_forbidden_knowledge_full_reward_effect = {
	if = {
		limit = {
			has_trait = witch
			has_trait_xp = { trait = witch value >= 100 }
		}
		witch_improved_add_current_lifestyle_perk_effect = { POINTS = 2 }
	}
	else = {
		witch_improved_add_current_lifestyle_perk_effect = { POINTS = 1 }
		witch_improved_add_current_lifestyle_xp_effect = { XP = 500 }
	}
}

# Seek Forbidden Knowledge, risky option failed: one grounded bad outcome, shown as a toast. Scope: the host.
# Permanent traits are rare; the strong brew makes them more likely.
witch_forbidden_knowledge_failure_effect = {
	random_list = {
		35 = {
			show_chance = no
			send_interface_toast = {
				title = witchimproved.1004.a.bad
				add_character_modifier = {
					modifier = witch_knowledge_sleepless_modifier
					years = 1
				}
			}
		}
		25 = {
			show_chance = no
			send_interface_toast = {
				title = witchimproved.1004.a.bad
				add_stress = major_stress_gain
			}
		}
		20 = {
			show_chance = no
			send_interface_toast = {
				title = witchimproved.1004.a.bad
				add_character_modifier = {
					modifier = witch_knowledge_muddled_modifier
					years = 2
				}
			}
		}
		15 = {
			trigger = { can_contract_disease_trigger = { DISEASE = ill } }
			show_chance = no
			send_interface_toast = {
				title = witchimproved.1004.a.bad
				contract_disease_effect = { DISEASE = ill TREATMENT_EVENT = yes }
			}
		}
		5 = {
			trigger = {
				NOR = {
					has_trait = paranoid
					has_trait = trusting
				}
			}
			modifier = {
				witch_improved_strong_brew_trigger = yes
				add = 5
			}
			show_chance = no
			send_interface_toast = {
				title = witchimproved.1004.a.bad
				add_trait = paranoid
			}
		}
		10 = {
			trigger = {
				witch_improved_strong_brew_trigger = yes
				NOT = { has_trait = lunatic }
			}
			show_chance = no
			send_interface_toast = {
				title = witchimproved.1004.a.bad
				add_trait = lunatic_1
			}
		}
	}
}
