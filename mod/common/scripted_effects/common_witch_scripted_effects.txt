
#Grand rite choice
grand_rite_good_outcome = {
	scope:activity = {
		every_attending_character = {
			limit = { this != root }
			custom = every_grand_rite_participant
			add_opinion = {
				target = root
				modifier = impressed_opinion
				opinion = grand_rite_good_outcome_opinion_value
			}
		}
	}
}

grand_rite_poor_outcome = {
	scope:activity = {
		every_attending_character = {
			limit = { this != root }
			custom = every_grand_rite_participant
			add_opinion = {
				target = root
				modifier = disappointed_opinion
				opinion = grand_rite_bad_outcome_opinion_value
			}
		}
	}
	add_stress = minor_stress_gain
}

hosted_successful_witch_coven_new_effect = {
	# Piety gain from Tenets.
	if = {
		limit = {
			rite = { rite_has_parameter = piety_from_witch_coven_active }
		}
		add_piety = grand_rite_piety_reward_value
	}

	# prestige gain from culture
	if = {
		limit = {
			culture = { has_cultural_parameter = witch_trait_more_valued }
		}
		add_prestige = grand_rite_prestige_reward_value
	}

	witch_improved_witchcraft_trait_xp_gain_effect = yes
}

attended_successful_witch_coven_new_effect = {# witch trait xp gain
	witch_improved_witchcraft_trait_xp_gain_effect = yes
}

disburse_witch_ritual_participant_rewards_effect = {
	save_temporary_scope_as = participant
	scope:activity = {
		add_activity_log_entry = {
			key = host_witch_ritual
			tags = { completed }
			# this line below adds the entry to the Effects section of the conclusion UI
			show_in_conclusion = yes
			character = scope:participant

			scope:participant = {
				attended_successful_witch_coven_new_effect = yes
			}
		}
	}
}

disburse_witch_ritual_host_rewards_effect = {
	scope:activity = {
		add_activity_log_entry = {
			key = host_witch_ritual
			tags = { completed }
			# this line below adds the entry to the Effects section of the conclusion UI
			show_in_conclusion = yes

			scope:host = {
				add_stress = grand_rite_host_stress_loss_value

		}

	add_trait_xp = {
		trait = witch
		value = {
			# from age 16-66 potential of 10 grand rites with avrg of 7.5, you can also attend others grand rites as well
			integer_range = {
				min = 5

			}
		}
	}
}

# Brew option of the Grand Rite (category rite_option_brew).
# Scope: the host. Run from the ritual phase on_end, so scope:activity and scope:host exist.
# Saves scope:brew_helper (an attending herbalist or court physician, if any) for the event portraits.
witch_improved_brew_outcome_effect = {
	scope:activity = {
		random_attending_character = {
			limit = {
				NOT = { this = scope:host }
				witch_improved_brew_helper_trigger = yes
			}
			save_scope_as = brew_helper
		}
	}
	if = {
		limit = {
			scope:activity = {
				has_activity_option = { category = rite_option_brew option = rite_brew_tea }
			}
		}
		trigger_event = witchimproved.1010
	}
	else_if = {
		limit = {
			scope:activity = {
				has_activity_option = { category = rite_option_brew option = rite_brew_strong }
			}
		}
		# Risky: a sick host can be cured, get a small boost, or get worse. A healthy host can also have a bad trip.
		if = {
			limit = { has_treatable_disease_trigger = yes }
			random_list = {
				40 = { trigger_event = witchimproved.1011 } # cured
				30 = { trigger_event = witchimproved.1013 } # not cured, small health boost
				30 = { trigger_event = witchimproved.1014 } # symptoms worsen
			}
		}
		else = {
			random_list = {
				85 = { trigger_event = witchimproved.1012 } # medium health boost
				15 = { trigger_event = witchimproved.1014 } # bad trip
			}
		}
	}
}
