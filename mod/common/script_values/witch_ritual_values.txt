# Seek Forbidden Knowledge: target value of the learning duel in witchimproved.1004.
# Scope: the host. Reads scope:grand_rite_ceremony_outcome and scope:activity when they exist.
witch_forbidden_knowledge_difficulty_value = {
	value = very_high_skill_rating
	# Experienced witches read the old texts more easily
	if = {
		limit = {
			has_trait = witch
			has_trait_xp = { trait = witch value >= 100 }
		}
		subtract = 4
	}
	else_if = {
		limit = {
			has_trait = witch
			has_trait_xp = { trait = witch value >= 50 }
		}
		subtract = 2
	}
	# A rite that went well steadies the mind, a botched one unsettles it
	if = {
		limit = { witch_improved_good_ceremony_trigger = yes }
		subtract = 2
	}
	else_if = {
		limit = { witch_improved_poor_ceremony_trigger = yes }
		add = 2
	}
	# The strong brew opens the mind (and makes a failure worse, see witch_forbidden_knowledge_failure_effect)
	if = {
		limit = { witch_improved_strong_brew_trigger = yes }
		subtract = 2
	}
	min = 8
}

# Chance (0-100) that the victim learns who was behind the hex. Scope: the hexer, in witchimproved.1003.
# Reads scope:target, scope:hex_type, scope:hex_result and scope:hex_helper.
witch_hex_discovery_chance_value = {
	value = 25
	subtract = intrigue
	# A sharp victim asks the right questions
	if = {
		limit = { scope:target = { intrigue >= high_skill_rating } }
		add = 10
	}
	# A botched hex leaves traces
	if = {
		limit = { NOT = { scope:hex_result = flag:success } }
		add = 15
	}
	if = {
		limit = { scope:hex_type = flag:pestilence }
		add = 5
	}
	# A skilled hand hides the poison
	if = {
		limit = {
			scope:hex_type = flag:sickness
			exists = scope:hex_helper
		}
		subtract = 5
	}
	min = 5
	max = 60
}

# The value of the character's highest skill (prowess excluded). Scope: character

	value = 0

	}
	else_if = {
		limit = { highest_skill = martial }
		add = martial
	}
	else_if = {
		limit = { highest_skill = stewardship }
		add = stewardship
	}
	else_if = {
		limit = { highest_skill = intrigue }
		add = intrigue
	}
	else_if = {
		limit = { highest_skill = learning }
		add = learning
	}
}
