# Overrides of the vanilla witchcraft doctrines (20_doctrines.txt) plus the new "worshipped" doctrine.
# 1.20 format: each doctrine names its group (doctrine_group_type) and index; piety_cost root scope is a Rite.
# Opinion effects between faiths use special_parameters.parameter_opinion_override: the key is a doctrine
# parameter that the OTHER faith has, the value is the opinion this faith's characters get of them.

doctrine_witchcraft_crime = {
	doctrine_group_type = doctrine_witchcraft
	index = 0
	divergence = 0

	piety_cost = {
		value = faith_doctrine_cost_low
		if = {
			limit = { rite_has_doctrine = doctrine_witchcraft_crime }
			multiply = faith_unchanged_doctrine_cost_mult
		}
		else_if = {
			limit = { rite_has_doctrine = doctrine_witchcraft_accepted }
			multiply = faith_changed_doctrine_cost_mult_two_step
		}
		else_if = {
			limit = { rite_has_doctrine = doctrine_witchcraft_virtuous }
			multiply = faith_changed_doctrine_cost_mult_three_step
		}
		else_if = {
			limit = { rite_has_doctrine = doctrine_witchcraft_worshipped }
			multiply = faith_changed_doctrine_cost_mult_four_step
		}
	}

	parameters = {
		witchcraft_illegal
		#witchcraft_rejecting_faiths
	}

	traits = {
		sins = {
			witch = { weight = 80 scale = 2 }
		}
	}
}

	index = 4
	divergence = 25

	piety_cost = {
